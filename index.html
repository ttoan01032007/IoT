2. Mã nguồn mẫu cho ESP32-S3 (Arduino IDE)

/*
 * Smart Aquarium Control - ESP32-S3
 * Broker: broker.emqx.io (Port 1883 TCP hoặc 8084 WSS)
 */
#include <WiFi.h>
#include <PubSubClient.h>

const char* ssid = "YOUR_WIFI";
const char* password = "YOUR_PASSWORD";
const char* mqtt_server = "broker.emqx.io";

const int PIN_PHAO = 11;
const int PIN_PUMP = 4;
const int PIN_LED = 5;
const int PIN_FAN = 6;
const int PIN_CHILLER = 7;

WiFiClient espClient;
PubSubClient client(espClient);
int lastWaterState = -1;

void callback(char* topic, byte* payload, unsigned int length) {
  String msg = "";
  for (int i = 0; i < length; i++) msg += (char)payload[i];
  String t = String(topic);

  if (t.endsWith("/pump")) digitalWrite(PIN_PUMP, msg == "on" ? HIGH : LOW);
  else if (t.endsWith("/led")) digitalWrite(PIN_LED, msg == "on" ? HIGH : LOW);
  else if (t.endsWith("/fan")) digitalWrite(PIN_FAN, msg == "on" ? HIGH : LOW);
  else if (t.endsWith("/chiller")) digitalWrite(PIN_CHILLER, msg == "on" ? HIGH : LOW);
}

void reconnect() {
  while (!client.connected()) {
    String clientId = "ESP32S3_" + String(random(0xffff), HEX);
    if (client.connect(clientId.c_str())) {
      client.subscribe("smart_aquarium/esp32s3/pump");
      client.subscribe("smart_aquarium/esp32s3/led");
      client.subscribe("smart_aquarium/esp32s3/fan");
      client.subscribe("smart_aquarium/esp32s3/chiller");
    } else {
      delay(3000);
    }
  }
}

void setup() {
  pinMode(PIN_PHAO, INPUT_PULLUP);
  pinMode(PIN_PUMP, OUTPUT);
  pinMode(PIN_LED, OUTPUT);
  pinMode(PIN_FAN, OUTPUT);
  pinMode(PIN_CHILLER, OUTPUT);

  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) delay(500);

  client.setServer(mqtt_server, 1883);
  client.setCallback(callback);
}

void loop() {
  if (!client.connected()) reconnect();
  client.loop();

  int state = digitalRead(PIN_PHAO);
  if (state != lastWaterState) {
    lastWaterState = state;
    client.publish("smart_aquarium/esp32s3/muc_nuoc", state == HIGH ? "1" : "0");
  }
  delay(150);
}
